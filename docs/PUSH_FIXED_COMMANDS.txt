# ================================================================

# Copy và chạy từng lệnh một ở máy công ty
# ================================================================

# 1. Vào thư mục (sửa đường dẫn cho đúng)
cd D:\ERP-Carton

# 2. Kiểm tra branch
git branch

# 3. Đảm bảo ở đúng branch
git checkout sprint2-business-models

# 4. Pull code mới nhất từ remote (QUAN TRỌNG!)
git pull origin sprint2-business-models

# 5. Nếu có conflict, xem file FIX_PUSH_ERROR.md để xử lý

# 6. Add tất cả file
git add .

# 7. Commit
git commit -m "feat: Sync all changes from company machine"

# 8. Push lại
git push origin sprint2-business-models

# 9. Kiểm tra kết quả
git status
